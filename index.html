/* ========================================================= REVEZAMENTO INTELIGENTE - PESCA JUSTA ========================================================= */ function embaralhar(lista) { const copia = [...lista]; for (let i = copia.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); [copia[i], copia[j]] = [ copia[j], copia[i] ]; } return copia; } function gerarCronogramaTurnos() { /* ======================================================= 1 - PREPARAR PARTICIPANTES ======================================================= */ const nomes = [ ...new Set( state.participants .map(p => (p.name || '').trim()) .filter(nome => nome !== '') ) ]; if (nomes.length === 0) { return toast('Cadastre pescadores primeiro.'); } /* ======================================================= 2 - CONFIGURAÇÕES DE TEMPO ======================================================= */ const horasPorTurno = parseFloat( document.getElementById( 'horasPorTurnoInput' ).value ) || 4; const inicioStr = document.getElementById( 'inicioPescaInput' ).value; const fimStr = document.getElementById( 'fimPescaInput' ).value; const inicioMs = new Date(inicioStr).getTime(); const fimMs = new Date(fimStr).getTime(); if (fimMs <= inicioMs) { return toast( 'A data/hora final deve ser posterior ao início.' ); } const duracaoTurnoMs = horasPorTurno * 60 * 60 * 1000; const duracaoTotalHoras = (fimMs - inicioMs) / (1000 * 60 * 60); const numTurnos = Math.ceil( duracaoTotalHoras / horasPorTurno ); /* ======================================================= 3 - CONFIGURAÇÃO DO BARCO ======================================================= */ const cfg = state.barcoConfig; const esqCount = cfg.bancosEsq * cfg.capEsq; const dirCount = cfg.bancosDir * cfg.capDir; const popaCount = cfg.bancosPopa * cfg.capPopa; const totalVagas = esqCount + dirCount + popaCount; if (totalVagas <= 0) { return toast( 'Configure corretamente as vagas do barco.' ); } if (nomes.length < totalVagas) { return toast( `Existem ${nomes.length} pescadores para ${totalVagas} vagas. ` + `Cadastre mais pescadores ou reduza as vagas.` ); } /* ======================================================= 4 - ESTATÍSTICAS DE CADA PESCADOR ======================================================= */ const estatisticas = {}; nomes.forEach(nome => { estatisticas[nome] = { // Quantidade total de turnos pescados turnos: 0, // Quantidade de vezes em cada área esquerda: 0, direita: 0, popa: 0, // Último turno em que participou ultimoTurno: -999, // Última posição ultimaArea: null, // Quantidade de descansos descansos: 0 }; }); /* Embaralha uma única vez para que, quando houver empate entre pescadores, não seja sempre o primeiro nome da lista a receber prioridade. */ const ordemInicial = embaralhar(nomes); /* Número aleatório para desempate. */ const prioridadeAleatoria = {}; ordemInicial.forEach( (nome, index) => { prioridadeAleatoria[nome] = index; } ); /* ======================================================= 5 - INICIAR CRONOGRAMA ======================================================= */ state.cronograma = []; let horaAtual = new Date(inicioMs); let participantesTurnoAnterior = []; /* ======================================================= 6 - GERAR CADA TURNO ======================================================= */ for ( let turnoAtual = 1; turnoAtual <= numTurnos; turnoAtual++ ) { /* ===================================================== CALCULAR HORÁRIO DO TURNO ===================================================== */ let proximaHoraMs = horaAtual.getTime() + duracaoTurnoMs; if ( proximaHoraMs > fimMs ) { proximaHoraMs = fimMs; } const horaFim = new Date( proximaHoraMs ); /* ===================================================== IDENTIFICAR QUEM DESCANSOU ===================================================== */ const descansaram = nomes.filter( nome => !participantesTurnoAnterior .includes(nome) ); /* A partir do segundo turno, quem descansou recebe prioridade. */ /* ===================================================== RANKING INTELIGENTE ===================================================== */ const ranking = [...nomes] .sort( (a, b) => { const ea = estatisticas[a]; const eb = estatisticas[b]; /* --------------------------------------------- PRIORIDADE 1 QUEM PESCOU MENOS TURNOS --------------------------------------------- */ if ( ea.turnos !== eb.turnos ) { return ( ea.turnos - eb.turnos ); } /* --------------------------------------------- PRIORIDADE 2 QUEM DESCANSOU NO TURNO ANTERIOR --------------------------------------------- */ const aDescansou = participantesTurnoAnterior.length > 0 && !participantesTurnoAnterior.includes(a); const bDescansou = participantesTurnoAnterior.length > 0 && !participantesTurnoAnterior.includes(b); if ( aDescansou !== bDescansou ) { return aDescansou ? -1 : 1; } /* --------------------------------------------- PRIORIDADE 3 QUEM ESTÁ HÁ MAIS TEMPO SEM PESCAR --------------------------------------------- */ if ( ea.ultimoTurno !== eb.ultimoTurno ) { return ( ea.ultimoTurno - eb.ultimoTurno ); } /* --------------------------------------------- PRIORIDADE 4 DESEMPATE ALEATÓRIO FIXO --------------------------------------------- */ return ( prioridadeAleatoria[a] - prioridadeAleatoria[b] ); } ); /* Seleciona somente a quantidade necessária para preencher o barco. */ const pescadoresTurno = ranking.slice( 0, totalVagas ); /* ===================================================== 7 - CONTROLE DOS DESCANSOS ===================================================== */ nomes.forEach( nome => { if ( !pescadoresTurno.includes(nome) ) { estatisticas[nome] .descansos++; } } ); /* ===================================================== 8 - DISTRIBUIÇÃO INTELIGENTE NAS ÁREAS ===================================================== */ const pescadoresDisponiveis = [...pescadoresTurno]; /* Função que escolhe pescadores tentando evitar repetir a mesma área. */ function escolherParaArea( area, quantidade ) { if ( quantidade <= 0 ) { return []; } const ordenados = pescadoresDisponiveis .sort( (a, b) => { const ea = estatisticas[a]; const eb = estatisticas[b]; /* Quantidade de vezes que já ficou nesta área */ const vezesA = ea[area]; const vezesB = eb[area]; if ( vezesA !== vezesB ) { return ( vezesA - vezesB ); } /* Evita repetir a mesma área do turno anterior */ const aMesmaArea = ea.ultimaArea === area ? 1 : 0; const bMesmaArea = eb.ultimaArea === area ? 1 : 0; if ( aMesmaArea !== bMesmaArea ) { return ( aMesmaArea - bMesmaArea ); } /* Quem pescou menos tem prioridade */ if ( ea.turnos !== eb.turnos ) { return ( ea.turnos - eb.turnos ); } return ( prioridadeAleatoria[a] - prioridadeAleatoria[b] ); } ); const escolhidos = ordenados.slice( 0, quantidade ); /* Remove os escolhidos da lista disponível. */ escolhidos.forEach( nome => { const index = pescadoresDisponiveis .indexOf(nome); if ( index >= 0 ) { pescadoresDisponiveis .splice( index, 1 ); } } ); /* Atualiza estatísticas da área. */ escolhidos.forEach( nome => { estatisticas[nome] [area]++; estatisticas[nome] .ultimaArea = area; } ); return escolhidos; } /* Distribuição das áreas. */ const pEsq = escolherParaArea( 'esquerda', esqCount ); const pDir = escolherParaArea( 'direita', dirCount ); const pPopa = escolherParaArea( 'popa', popaCount ); /* ===================================================== 9 - ATUALIZAR ESTATÍSTICAS DE PARTICIPAÇÃO ===================================================== */ pescadoresTurno.forEach( nome => { estatisticas[nome] .turnos++; estatisticas[nome] .ultimoTurno = turnoAtual; } ); /* ===================================================== 10 - SALVAR TURNO ===================================================== */ state.cronograma.push({ turno: turnoAtual, inicio: new Date( horaAtual ), fim: new Date( horaFim ), esq: pEsq, dir: pDir, popa: pPopa, descanso: nomes.filter( nome => !pescadoresTurno .includes(nome) ) }); /* Guarda participantes para o próximo turno. */ participantesTurnoAnterior = [...pescadoresTurno]; horaAtual = horaFim; if ( horaAtual.getTime() >= fimMs ) { break; } } /* ======================================================= 11 - SALVAR E ATUALIZAR TELA ======================================================= */ save(); renderTabelaCronograma(); renderCardsTopoBarco(); toast( 'Revezamento inteligente gerado com sucesso! 🎣' ); }
